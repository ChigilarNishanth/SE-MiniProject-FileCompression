cmake_minimum_required(VERSION 3.16)
project(czip VERSION 1.0.0 LANGUAGES CXX)
add_executable(czip src/czip.cpp)
target_compile_features(czip PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(czip PRIVATE /W4 /WX)
else()
  target_compile_options(czip PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(CZIP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(CZIP_SANITIZE AND NOT MSVC)
  target_compile_options(czip PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(czip PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME requirements COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_czip.py $<TARGET_FILE:czip>)
